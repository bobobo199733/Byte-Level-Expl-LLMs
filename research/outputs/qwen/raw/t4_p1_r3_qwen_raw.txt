```python
def decode_string(s: str) -> str:
    stack = []
    current_num = 0
    current_str = ""
    
    for char in s:
        if char.isdigit():
            # Build the number
            current_num = current_num * 10 + int(char)
        elif char == '[':
            # Push the current string and number to the stack
            stack.append((current_str, current_num))
            current_str = ""
            current_num = 0
        elif char == ']':
            # Pop the last string and number from the stack
            last_str, num = stack.pop()
            current_str = last_str + current_str * num
        else:
            # Append the character to the current string
            current_str += char
    
    return current_str

# Example usage:
encoded_string = "3[a2[c]]"
decoded_string = decode_string(encoded_string)
print(decoded_string)  # Output: accaccacc

# Additional test cases
print(decode_string("3[abc]"))  # Output: abcabcabc
print(decode_string("2[3[a]b]"))  # Output: aaabaaab
```

This function `decode_string` takes a string `s` as input and returns the decoded string according to the specified rules. It uses a stack to handle nested patterns efficiently. The function validates the input by ensuring that all numbers are single-digit or multi-digit combinations, and it handles nested brackets appropriately.
