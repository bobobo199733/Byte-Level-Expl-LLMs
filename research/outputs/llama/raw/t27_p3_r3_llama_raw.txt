def trapping_rain_water(heights):
    """
    Calculate the amount of trapped rainwater in a given list of heights.

    Args:
        heights (list): A list of integers representing the height of each building.

    Returns:
        int: The total amount of trapped rainwater.
    """
    
    # Initialize two pointers, one at the start and one at the end
    left = 0
    right = len(heights) - 1
    
    # Initialize the maximum heights on the left and right sides
    max_left = 0
    max_right = 0
    
    # Initialize the total trapped rainwater
    total_water = 0
    
    # Iterate through the list from both ends
    while left < right:
        # If the height on the left is smaller, update the maximum height on the left and add to the total water
        if heights[left] < heights[right]:
            max_left = max(heights[left], max_left)
            total_water += max_left - heights[left]
            left += 1
        else:
            # Otherwise, update the maximum height on the right and add to the total water
            max_right = max(heights[right], max_right)
            total_water += max_right - heights[right]
            right -= 1
    
    return total_water
