```python
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def merge_k_sorted_lists(lists):
    # Helper function to merge two sorted linked lists
    def merge_two_lists(l1, l2):
        dummy = ListNode()
        current = dummy
        
        while l1 and l2:
            if l1.val < l2.val:
                current.next = l1
                l1 = l1.next
            else:
                current.next = l2
                l2 = l2.next
            current = current.next
        
        # Attach the remaining elements of l1 or l2
        current.next = l1 if l1 else l2
        
        return dummy.next
    
    while len(lists) > 1:
        merged_lists = []
        
        for i in range(0, len(lists), 2):
            l1 = lists[i]
            l2 = lists[i + 1] if (i + 1) < len(lists) else None
            merged_lists.append(merge_two_lists(l1, l2))
        
        lists = merged_lists
    
    return lists[0] if lists else None

# Helper function to print the linked list for testing purposes
def print_linked_list(node):
    while node:
        print(node.val, end=" -> ")
        node = node.next
    print("None")

# Example usage:
# Creating sample sorted lists
l1 = ListNode(1, ListNode(4, ListNode(5)))
l2 = ListNode(1, ListNode(3, ListNode(4)))
l3 = ListNode(2, ListNode(6))

lists = [l1, l2, l3]

result = merge_k_sorted_lists(lists)
```

### Explanation:
1. **ListNode Class**: Defines the structure of a node in the linked list.
2. **merge_two_lists Function**: Merges two sorted linked lists into one.
3. **merge_k_sorted_lists Function**:

   - Iterates over the list of lists, merging them in pairs and storing the results back in the `lists` array.
   - Continues this process until only one merged list remains.
