```python
def decode_string(s: str) -> str:
    stack = []
    current_number = 0
    current_string = ''
    
    for char in s:
        if char.isdigit():
            # Build the number by concatenating digits
            current_number = current_number * 10 + int(char)
        elif char.isalpha():
            # Append alphabet character to the current string
            current_string += char
        elif char == '[':
            # Push the current string and number to stack if they exist
            stack.append((current_string, current_number))
            # Reset current string and number
            current_string = ''
            current_number = 0
        elif char == ']':
            # Pop the last string and number from the stack
            prev_string, num = stack.pop()
            # Concatenate the current string with the previous one and repeat as per the number
            current_string = prev_string + num * current_string
    
    return current_string

# Example usage:
print(decode_string("3[a]2[bc]") == "aaabcbc")
print(decode_string("3[a2[c]]") == "accaccacc")
print(decode_string("2[abc]3[cd]ef") == "abcabccdcdcdef")
```

This function `decode_string` takes a string `s` as input and returns the decoded string according to the specified rules. It uses a stack to handle nested patterns and correctly reconstructs the original string by following the encoded structure.
