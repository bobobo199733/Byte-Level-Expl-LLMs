def robot_simulator(commands):
    x, y = 0, 0  # Starting point at the origin
    direction = 0  # Initial direction: 0=North, 1=East, 2=South, 3=West
    
    for command in commands:
        if command == 'G':
            if direction == 0: y += 1
            elif direction == 1: x += 1
            elif direction == 2: y -= 1
            elif direction == 3: x -= 1
        elif command == 'L':
            direction = (direction + 3) % 4  # Turn left (subtract 90 degrees)
        elif command == 'R':
            direction = (direction + 1) % 4   # Turn right (add 90 degrees)
    
    return x, y
