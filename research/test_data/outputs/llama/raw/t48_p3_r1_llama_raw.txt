```python
def median_of_two_sorted_arrays(nums1, nums2):
    """
    Calculate the median of two sorted arrays efficiently using binary search.

    Args:
        nums1 (list): The first sorted array.
        nums2 (list): The second sorted array.

    Returns:
        float: The median of the combined arrays.
    """

    # Make sure that nums1 is the smaller array
    if len(nums1) > len(nums2):
        nums1, nums2 = nums2, nums1

    total_length = len(nums1) + len(nums2)
    half_length = total_length // 2

    left, right = 0, len(nums1) - 1

    while True:
        # Calculate the partition points for both arrays
        i = (left + right) // 2
        j = half_length - i - 2

        # Partition nums1
        nums1_left = nums1[i + 1] if i < len(nums1) else float('-infinity')
        nums1_right = nums1[i] if i >= 0 else float('inf')

        # Partition nums2
        nums2_left = nums2[j + 1] if j < len(nums2) else float('-infinity')
        nums2_right = nums2[j] if j >= 0 else float('inf')

        # Check if the partition is correct
        if nums1_left <= nums2_right and nums2_left <= nums1_right:
            # If the total length is even, return the average of the two middle numbers
            if total_length % 2 == 0:
                return (max(nums1_left, nums2_left) + min(nums1_right, nums2_right)) / 2.0

            # Otherwise, return the larger of the two middle numbers
            else:
                return max(min(nums1_right, nums2_right), max(nums1_left, nums2_left))
        # If the partition is not correct, adjust it and try again
        elif nums1_left > nums2_right:
            right = i - 1
        else:
            left = i + 1

print(median_of_two_sorted_arrays([1, 3], [2]))  # Output: 2.0
```
