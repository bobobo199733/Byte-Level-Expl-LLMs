Here's the function `sudoku_solver` to solve a standard 9x9 Sudoku puzzle:

```python
def sudoku_solver(board):
    """
    Solves a standard 9x9 Sudoku puzzle using backtracking.

    Args:
        board (list): A 2D list representing the Sudoku puzzle. Empty cells are represented by 0.
            The puzzle must be valid and solvable for this function to work.

    Raises:
        ValueError: If the input puzzle is invalid or unsolvable.

    Returns:
        None
    """
    def is_valid(board, row, col, num):
        # Check if the number already exists in the current box (3x3)
        box_size = 9 // 3
        box_row_start, box_col_start = row - row % box_size, col - col % box_size
        for r in range(box_row_start, box_row_start + box_size):
            for c in range(box_col_start, box_col_start + box_size):
                if board[r][c] == num:
                    return False

        # Check if the number already exists in the row or column
        if num in board[row]:
            return False
        if num in [board[r][col] for r in range(9)]:
            return False

        return True

    def fill_board(board):
        """
        Attempts to fill a cell with a valid number using backtracking.
        If no valid number can be found, it raises an exception.

        Args:
            board (list): A 2D list representing the Sudoku puzzle.

        Returns:
            bool: Whether or not all cells were filled successfully.
        """
        for row in range(9):
            for col in range(9):
                if board[row][col] == 0:
                    # Try each possible number
                    for num in range(1, 10):
                        if is_valid(board, row, col, num):
                            # Fill the cell with a valid number and recurse
                            board[row][col] = num
                            if fill_board(board):
                                return True
                            # If backtracking fails, reset the cell to empty and try another number
                            board[row][col] = 0
                    raise ValueError("Puzzle is unsolvable")
        return True

    try:
        fill_board(board)
    except ValueError as e:
        raise ValueError(f"Invalid or unsolvable puzzle: {e}")
```

Here's an example use case to demonstrate the function:

```python
# Example Sudoku puzzle
board = [
    [5, 3, 0, 0, 7, 0, 0, 0, 0],
    [6, 0, 0, 1, 9, 5, 0, 0, 0],
    [0, 9, 8, 0, 0, 0, 0, 6, 0],
    [8, 0, 0, 0, 6, 0, 0, 0, 3],
    [4, 0, 0, 8, 0, 3, 0, 0, 1],
    [7, 0, 0, 0, 2, 0, 0, 0, 6],
    [0, 6, 0, 0, 0, 0, 2, 8, 0],
    [0, 0, 0, 4, 1, 9, 0, 0, 5],
    [0, 0, 0, 0, 8, 0, 0, 7, 9]
]

sudoku_solver(board)

# The filled board is now available
print(board)
```

This will fill the Sudoku puzzle and print the solved board.
