```python
def roman_to_int(s: str) -> int:
    """
    Converts a Roman numeral string into its integer value.
    
    Args:
        s (str): A string representing a Roman numeral, using standard symbols I, V, X, L, C, D, M.
        
    Returns:
        int: The integer equivalent of the input Roman numeral string.
        
    Raises:
        ValueError: If the input contains invalid characters or if subtractive notation is used incorrectly.
    """

    # Define a dictionary to map each symbol to its corresponding integer value
    roman_numerals = {
        'I': 1, 'V': 5, 'X': 10, 'L': 50, 'C': 100, 'D': 500, 'M': 1000
    }

    # Initialize the result and the previous symbol's value
    result = 0
    prev_value = 0

    # Iterate over each character in the input string
    for numeral in s:
        if numeral not in roman_numerals:
            raise ValueError(f"Invalid Roman numeral: {numeral}")

        # If the current symbol is greater than or equal to the previous one, add its value
        current_value = roman_numerals[numeral]
        if current_value >= prev_value:
            result += current_value
        else:
            # Otherwise, subtract twice the current symbol's value and add its value (due to subtractive notation)
            raise ValueError(f"Invalid subtractive notation: {numeral}")

        # Update the previous symbol's value
        prev_value = current_value

    return result
```

This function assumes that the input string only contains valid Roman numerals and handles both simple addition and subtractive notation cases correctly.
