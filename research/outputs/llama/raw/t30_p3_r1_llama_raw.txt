def max_subarray_sum(nums):
    """
    This function calculates the maximum sum of a contiguous subarray.

    Args:
        nums (list): A list of integers.

    Returns:
        int: The maximum sum of a contiguous subarray.
    """

    # Handle edge case where input is empty
    if not nums:
        return 0

    # Initialize current max and overall max to the first element in the array
    curr_max = overall_max = nums[0]

    # Iterate over the rest of the array
    for num in nums[1:]:
        
        # Update the current max by adding the current number, or reset it if it's smaller
        curr_max = max(num, curr_max + num)
        
        # Update the overall max if the current max is larger
        overall_max = max(overall_max, curr_max)

    return overall_max
