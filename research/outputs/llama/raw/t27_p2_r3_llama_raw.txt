def trapping_rain_water(heights):
    """
    Computes the total trapped rain water in a given list of bar heights.

    Args:
        heights (list): A list of non-negative integers representing bar heights.

    Returns:
        int: The total amount of trapped rain water.
    """
    if not heights:  # handle empty lists
        return 0
    
    n = len(heights)
    
    left_max = [0] * n  # max height on the left side up to each index
    right_max = [0] * n  # max height on the right side up to each index
    
    left_max[0] = heights[0]
    for i in range(1, n):
        left_max[i] = max(left_max[i - 1], heights[i])
    
    right_max[n - 1] = heights[n - 1]
    for i in range(n - 2, -1, -1):
        right_max[i] = max(right_max[i + 1], heights[i])
    
    trapped_water = 0
    for i in range(n):
        trapped_water += min(left_max[i], right_max[i]) - heights[i]
    
    return trapped_water
