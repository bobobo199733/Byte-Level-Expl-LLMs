def merge_intervals(intervals):
    """
    Merges overlapping intervals in a given list of intervals.

    Args:
        intervals (list): A list of intervals where each interval is a list of two integers.

    Returns:
        list: A list of merged intervals.
    """

    # Sort the intervals based on their start value
    intervals.sort(key=lambda x: x[0])

    # Initialize the result with the first interval
    merged = [intervals[0]]

    # Iterate over the remaining intervals
    for current in intervals[1:]:
        # Get the last merged interval
        last_merged = merged[-1]

        # Check if the current interval overlaps with the last merged interval
        if current[0] <= last_merged[1]:
            # Merge the two intervals by updating the end value of the last merged interval
            last_merged[1] = max(last_merged[1], current[1])
        else:
            # Add the current interval to the result if it does not overlap with the last merged interval
            merged.append(current)

    return merged
