Here's an implementation of the Sudoku solver in Python using backtracking algorithm:

```python
def is_valid(row, col, num, board):
    for i in range(9):
        if board[row][i] == num or board[i][col] == num:
            return False

    box_x = row - (row % 3)
    box_y = col - (col % 3)
    for i in range(box_x, box_x + 3):
        for j in range(box_y, box_y + 3):
            if board[i][j] == num:
                return False
    return True

def sudoku_solver(board):
    empty_cells = find_empty_cells(board)

    for row, col in empty_cells:
        for num in range(1, 10):
            if is_valid(row, col, num, board):
                board[row][col] = num
                if sudoku_solver(board):
                    return True
            board[row][col] = 0  # backtracking
    return False

def find_empty_cells(board):
    for row in range(9):
        for col in range(9):
            if board[row][col] == 0:
                yield row, col

def print_board(board):
    for row in board:
        for num in row:
            print(num, end=' ')
        print()
    print()

if __name__ == "__main__":
    board = [[3, 0, 6, 5, 0, 8, 4, 0, 0],
             [5, 2, 0, 0, 0, 0, 0, 0, 0],
             [0, 8, 7, 0, 0, 0, 0, 3, 1],
             [0, 0, 3, 0, 1, 0, 0, 8, 0],
             [9, 0, 0, 8, 6, 3, 0, 0, 5],
             [0, 5, 0, 0, 9, 0, 7, 6, 0],
             [1, 3, 0, 0, 0, 0, 2, 5, 0],
             [0, 0, 4, 7, 0, 5, 3, 0, 6],
             [7, 0, 1, 0, 0, 2, 0, 0, 8]]

    print_board(board)
    sudoku_solver(board)
    print_board(board)
```

This code defines a helper function `is_valid` to check if a given number can be placed in a specific cell without violating any Sudoku rules, and another helper function `find_empty_cells` to find the empty cells on the board.

The main function `sudoku_solver` uses backtracking to solve the Sudoku puzzle by iterating through each empty cell and trying all possible numbers (1-9) for that cell, recursively solving the subproblem if a valid solution is found. If no solution can be found, it backtracks by setting the number back to 0 before moving on to the next cell.
