Here's a Python function named `text_justification` that implements the requirements:

```python
def text_justification(words, max_width):
    if not words:
        return []

    lines = []
    current_line = []
    spaces_remaining = max_width

    for word in words:
        # Calculate the length of the word and the number of spaces to be added
        word_length = len(word)
        spaces_to_add = min(spaces_remaining, len(current_line) - 1) * ' '

        # Add the word and spaces to the current line if it doesn't exceed max_width
        if sum([word_length] + list(map(len, current_line))) + spaces_to_add <= max_width:
            current_line.append(word + spaces_to_add)
            spaces_remaining -= word_length + spaces_to_add
        # If the line is full or exceeds max_width, create a new line and add the word with proper spacing
        else:
            lines.append(' '.join(current_line))
            current_line = [word]
            spaces_remaining -= word_length + 1

    # Add any remaining words in the last line
    lines.append(' '.join(current_line))

    # If there's a single-word last line, pad it with trailing spaces to reach max_width
    if len(lines) > 1 and len(lines[-1]) < max_width:
        lines[-1] += ' ' * (max_width - len(lines[-1]))

    return lines
```

You can use the function like this:

```python
text = "Lorem ipsum dolor sit amet, consectetur adipiscing elit. Maecenas feugiat vestibulum augue, in faucibus nulla egestas a."
max_width = 30
print(text_justification(text.split(), max_width))
```
