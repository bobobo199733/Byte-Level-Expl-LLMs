```python
def expression_to_postfix(expression):
    # Operator precedence dictionary
    precedence = {'+': 1, '-': 1, '*': 2, '/': 2, '^': 3}
    
    def is_operator(char):
        return char in '+-*/^()'
    
    def to_token(value):
        if value.isalpha() or value.isdigit():
            return 'value', value
        else:
            return 'operator', value
    
    def higher_precedence(op1, op2):
        return precedence[op1] >= precedence[op2]
    
    def apply_operator(operators, values):
        operator = operators.pop()
        right = values.pop()
        left = values.pop()
        values.append(f'({left} {operator} {right})')
    
    tokens = [to_token(char) for char in expression if not char.isspace()]
    operators_stack = []
    output_queue = []
    
    for token in tokens:
        if token[0] == 'value':
            output_queue.append(token[1])
        elif is_operator(token[1]):
            while (operators_stack and 
                   higher_precedence(operators_stack[-1][1], token[1]) and
                   operators_stack[-1][1] != '('):
                apply_operator(operators_stack, output_queue)
            operators_stack.append(token)
        elif token[1] == '(':
            operators_stack.append(token)
        elif token[1] == ')':
            while operators_stack and operators_stack[-1][1] != '(':
                apply_operator(operators_stack, output_queue)
            if operators_stack and operators_stack[-1][1] == '(':
                operators_stack.pop()
    
    # Apply any remaining operators
    while operators_stack:
        apply_operator(operators_stack, output_queue)
        
    return ' '.join(output_queue)

# Example usage:
infix_expression = "3 + 4 * 2 / ( 1 - 5 ) ^ 2 ^ 3"
postfix_expression = expression_to_postfix(infix_expression)
print(postfix_expression)  # Output: 3 4 2 * 1 5 - 2 3 ^ ^ / +
```

### Explanation:
- **Precedence Dictionary**: The `precedence` dictionary defines the precedence of operators.
- **Tokenization**: The expression is split into tokens, where each token is either an operator or a value (operand).
- **Stacks**: 
  - `operators_stack`: Used to hold operators and manage their precedence.
  - `output_queue`: Holds the final postfix expression as it is built.
- **Operator Evaluation**:
  - If a token is a value, it's directly added to the output queue.
  - If a token is an operator, we check its precedence against the top of the stack. If it has higher or equal precedence, operators are popped from the stack and applied to values until we can add the current operator.
  - Parentheses (`(` and `)`) are used to manage the order of operations correctly.
- **Postfix Construction**: After processing all tokens, any remaining operators in the stack are applied to the output queue.

This implementation ensures that the given infix expression is converted correctly into its postfix form while respecting operator precedence and parentheses.
