def count_set_bits(n):
    """
    Counts the number of set (1) bits in a given integer.

    Args:
        n (int): The input integer.

    Returns:
        int: The number of set bits in n.
    """

    # Use the built-in bin function to convert n into a binary string
    binary_str = bin(n)[2:]
    
    # Count the number of '1' characters in the binary string
    count = binary_str.count('1')
    
    return count
