Here's a Python function called `spiral_matrix` that takes a 2D matrix as input and returns all its elements in spiral order:

```python
def spiral_matrix(matrix):
    if not matrix or not matrix:
        return []

    rows, cols = len(matrix), len(matrix[0])
    direction = 0  # 0 - right, 1 - down, 2 - left, 3 - up
    visited = [[False] * cols for _ in range(rows)]

    result = []
    x, y = 0, 0

    while len(result) < rows * cols:
        visited[x][y] = True
        result.append(matrix[x][y])

        next_x, next_y = x, y

        if direction == 0:  # right
            if y + 1 < cols and not visited[x][y + 1]:
                next_x, next_y = x, y + 1
            else:
                direction += 1

        elif direction == 1:  # down
            if x + 1 < rows and not visited[x + 1][y]:
                next_x, next_y = x + 1, y
            else:
                direction += 1

        elif direction == 2:  # left
            if y - 1 >= 0 and not visited[x][y - 1]:
                next_x, next_y = x, y - 1
            else:
                direction += 1

        elif direction == 3:  # up
            if x - 1 >= 0 and not visited[x - 1][y]:
                next_x, next_y = x - 1, y
            else:
                x, y = next_x, next_y
                direction = (direction + 1) % 4  # wrap around to start a new row or column

    return result
```

You can test the function with the following example:

```python
matrix = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 9]
]
print(spiral_matrix(matrix))  # Output: [1, 2, 3, 6, 9, 8, 7, 4, 5]
```
