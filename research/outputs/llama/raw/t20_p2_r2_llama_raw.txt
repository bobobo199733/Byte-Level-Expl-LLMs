import heapq

def merge_k_sorted_lists(lists):
    """
    Merge k sorted lists into a single sorted list.

    Args:
        lists (list): A list containing sorted integer lists.

    Returns:
        list: A single sorted list containing all elements from the input lists.
    """

    # Handle edge cases
    if not lists:
        return []

    # Use heapq.merge to merge multiple sorted inputs into a single sorted output
    merged_list = []
    for lst in lists:
        if lst:
            merged_list.extend(lst)

    # Convert list of lists to a single list and sort it
    merged_list.sort()

    return merged_list
