Here's a Python function called `merge_k_sorted_lists` that merges multiple sorted lists into one sorted list. This function uses a priority queue (heapq in Python) for efficient sorting:

```python
import heapq

def merge_k_sorted_lists(lists):
    if not lists:
        return []
    min_heap = []

    # Initialize the min-heap with the first element from each list
    for list_ in lists:
        if list_ and len(min_heap) < len(lists):
            heapq.heappush(min_heap, (list_[0], 0, list_))
        elif list_ and len(min_heap) == len(lists):
            compared = False
            for _ in range(len(min_heap)):
                item, index, current_list = min_heap[0]
                if item > list_[0]:
                    heapq.heappop(min_heap)
                    break
                else:
                    compared = True
            if not compared:
                heapq.heappushpop(min_heap, (list_[0], 0, list_))

    result = []
    while min_heap:
        item, index, current_list = heapq.heappop(min_heap)
        result.append(item)
        if current_list and len(current_list) > 1:
            next_item = current_list[1]
            heapq.heappush(min_heap, (next_item, index, current_list))
    return result
```

Example usage:

```python
lists = [
    [1, 3, 5],
    [2, 4, 6, 8],
    [0]
]
print(merge_k_sorted_lists(lists))  # Output: [0, 1, 2, 3, 4, 5, 6, 8]
```
