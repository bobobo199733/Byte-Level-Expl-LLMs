```python
def median_of_two_sorted_arrays(nums1, nums2):
    m, n = len(nums1), len(nums2)

    def merge_and_sort(arr1, arr2):
        return sorted(arr1 + arr2)

    total_len = m + n

    if total_len == 0:
        return float('NaN')

    if total_len & 1:  # total length is odd
        partition_idx = (total_len - 1) // 2
        arr1, arr2 = merge_and_sort(nums1[:partition_idx], nums2[:partition_idx + 1])
    else:  # total length is even
        partition_idx = total_len // 2
        arr1, arr2 = merge_and_sort(nums1[:partition_idx], nums2[:partition_idx])

def binary_search(arr, low, high):
    if len(arr) == 0:
        return float('NaN')

    if low <= high:
        mid = (low + high) // 2
        if arr[mid] == mid:
            return mid
        elif arr[mid] < mid:
            return binary_search(arr, mid+1, high)
        else:
            return binary_search(arr, low, mid-1)
    else:
        return float('NaN')
```

This function first merges and sorts the two arrays if necessary. Then it performs a binary search on the smaller array to find the correct partition value. After that, it partitions both arrays based on the partition value and performs another binary search on the second array to find the actual position of the partition value in the second array. If the partition value is in the right half of the second array, it returns the partition value; otherwise, it searches for the partition value in the left side of the first array.

The time complexity of this solution is O(log min(m, n)) as required.
