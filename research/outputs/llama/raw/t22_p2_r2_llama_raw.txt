def sieve_primes(n):
    """Returns a list of all prime numbers up to and including n."""
    
    # Handle edge case where n is less than 2
    if n < 2:
        return []

    # Initialize boolean array, prime, of size n+1
    prime = [True] * (n + 1)
    
    # 0 and 1 are not prime numbers
    prime[0] = prime[1] = False
    
    # Iterate over the boolean array from 2 to sqrt(n)
    for p in range(2, int(n ** 0.5) + 1):
        
        # If p is a prime number, mark its multiples as non-prime
        if prime[p]:
            for i in range(p * p, n + 1, p):
                prime[i] = False
    
    # Return a list of all indices where the value is True
    return [p for p in range(2, n + 1) if prime[p]]
