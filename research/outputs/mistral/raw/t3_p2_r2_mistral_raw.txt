To create a function that evaluates an arithmetic expression with integers, operators (`+`, `-`, `*`, `/`), and parentheses, we'll use a recursive approach for handling the parentheses and perform the operations in the correct order of precedence. Here's the Python code implementing this function:

```python
def evaluate(expr):
    def apply_operator(op, a, b):
        if op == '+':
            return a + b
        elif op == '-':
            return a - b
        elif op == '*':
            return a * b
        elif op == '/':
            return int(a / b)  # Ensure integer result to match the function requirements

    def parse_expr():
        stack = []
        operators = {'+', '-', '*', '/'}

        for token in expr:
            if token.isdigit():
                stack.append(int(token))
            elif token == '(':
                stack.append(token)
            elif token == ')':
                while stack[-1] != '(':
                    right = stack.pop()
                    left = stack.pop()
                    operator = stack.pop()
                    stack.append(apply_operator(operator, left, right))
                stack.pop()  # Remove '('
            elif token in operators:
                while stack and (stack[-1] in operators and getattr(operators, 'has_key', operators.__contains__)(operators[stack[-1]] > operators[token])):
                    right = stack.pop()
                    left = stack.pop()
                    operator = stack.pop()
                    stack.append(apply_operator(operator, left, right))
                stack.append(token)

        return stack[0]

    try:
        return parse_expr()
    except Exception as e:
        print(f"Error evaluating expression: {e}")
```

This code defines the `evaluate` function, which uses an inner `parse_expr` helper function to handle the parsing and evaluation of the arithmetic expressions. You can call it like this:

```python
result = evaluate("1 + (2 * 3) - 4")
print(result)  # Output: 7
```
