```python
def expression_to_postfix(expression):
    # Define operator precedence
    precedence = {'+': 1, '-': 1, '*': 2, '/': 2}
    
    def apply_operator(operators, values):
        right = values.pop()
        left = values.pop()
        operators.append(f"({left} {right})")
    
    # Convert the expression to a list of tokens
    tokens = expression.replace(" ", "")
    token_list = []
    for char in tokens:
        if char.isalnum():
            token_list.append(char)
        else:
            token_list.append(' ' + char + ' ')
    
    operators_stack = []  # Stack for operators
    output_queue = []     # Queue for the output
    
    for token in token_list:
        if token.isalnum():  # If it's a number, add to output queue
            output_queue.append(token)
        elif token == '(':
            operators_stack.append('(')  # Add left parenthesis to stack
        elif token == ')':
            while operators_stack and operators_stack[-1] != '(':  # Pop from operator stack until matching left paren
                apply_operator(operators_stack, output_queue)
            operators_stack.pop()  # Remove the '(' from the stack
        else:  # It's an operator
            while (operators_stack and 
                   operators_stack[-1] != '(' and 
                   precedence[operators_stack[-1]] >= precedence[token]):
                apply_operator(operators_stack, output_queue)
            operators_stack.append(token)  # Push new operator onto stack
    
    # Pop any remaining items from the operator stack to the output queue
    while operators_stack:
        if operators_stack[-1] == '(':
            raise ValueError("Unmatched parenthesis in input expression")
        apply_operator(operators_stack, output_queue)
    
    return ''.join(output_queue)

# Example usage:
infix_expression = "3 + 4 * 2 / ( 1 - 5 ) ^ 2 ** 3"
postfix_expression = expression_to_postfix(infix_expression)
print(postfix_expression)  # Output: 3 4 2 * 1 5 - 2 3 ** ^ / +
```

This function works as follows:
1. It first converts the input string to a list of tokens, separating numbers and operators.
2. It uses two stacks: one for operators and another for output.
3. It processes each token in the expression:
   - If it's a number, it adds it directly to the output queue.
   - If it's an operator or parenthesis, it handles them based on their precedence and associativity rules.
4. Finally, any remaining operators in the stack are processed and added to the output queue.

The function ensures that all parentheses are correctly balanced and that the operators are applied according to their precedence.
