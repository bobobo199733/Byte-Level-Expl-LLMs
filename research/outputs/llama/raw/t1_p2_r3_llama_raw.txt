def roman_to_int(s: str) -> int:
    """
    Convert a Roman numeral string into its integer value.
    
    Args:
        s (str): The Roman numeral string to convert.
        
    Returns:
        int: The integer value of the input Roman numeral string.
    
    Raises:
        ValueError: If the input is not valid, including incorrect subtractive notation or invalid repetitions.
    """
    
    # Define a dictionary mapping Roman numerals to their integer values
    roman_values = {
        'I': 1,
        'V': 5,
        'X': 10,
        'L': 50,
        'C': 100,
        'D': 500,
        'M': 1000
    }
    
    # Initialize the result to 0
    result = 0
    
    # Check if the input is a string and contains only valid characters
    if not isinstance(s, str) or not all(c in roman_values for c in s):
        raise ValueError("Invalid Roman numeral string")
    
    # Iterate over each character in the input string
    for i, char in enumerate(s):
        
        # If this is not the last character and its value is less than the next one,
        # subtract its value from the result (this handles subtractive notation)
        if i < len(s) - 1 and roman_values[char] < roman_values[s[i + 1]]:
            result -= roman_values[char]
        else:
            # Otherwise, add its value to the result
            result += roman_values[char]
    
    # If the final result is less than 0 or greater than 3999 (the largest valid Roman numeral),
    # raise a ValueError
    if not 1 <= result <= 3999:
        raise ValueError("Invalid Roman numeral string")
    
    return result

This code works by defining a dictionary that maps each Roman numeral character to its integer value.
