def decode_string(s):

            stack.append(count)

            stack.append(char)

            while stack and stack[-1] != '[':

            stack.pop()  # Remove the opening bracket
        elif char.isalpha():  # If it's a regular character, just append it to the result
            result += char

    # Add any remaining characters from the stack to the result

        if stack[-1] == '[':
            raise ValueError("Error in input string")
