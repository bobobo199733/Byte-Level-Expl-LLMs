def trapping_rain_water(heights):
    """
    Calculate the total amount of rain water that can be trapped between bars.

    Args:
        heights (list): A list of integers representing the height of each bar.

    Returns:
        int: The total amount of rain water that can be trapped.
    """

    # Initialize two pointers, one at the start and one at the end
    left = 0
    right = len(heights) - 1

    # Initialize the maximum heights on the left and right
    max_left = 0
    max_right = 0

    # Initialize the total water trapped
    total_water = 0

    while left <= right:
        if heights[left] < heights[right]:
            # If the current height is less than the max height on the right, update the max height on the left
            if heights[left] >= max_left:
                max_left = heights[left]
            else:
                # Add the water trapped between the two to the total
                total_water += (max_left - heights[left]) * 1
            left += 1
        else:
            # If the current height is less than or equal to the max height on the left, update the max height on the right
            if heights[right] >= max_right:
                max_right = heights[right]
            else:
                # Add the water trapped between the two to the total
                total_water += (max_right - heights[right]) * 1
            right -= 1

    return total_water
