def robot_simulator(commands):
    """
    Simulates a robot moving on a grid according to the given commands.

    Args:
        commands (list): A list of commands where each command is a string.
                         'L' turns left, 'R' turns right,
                         and 'F', '<', '>', '^', 'v' move forward, left, right, up, down respectively.

    Returns:
        tuple: The final position and direction of the robot as (x, y, direction).
               x and y are integers representing the horizontal and vertical coordinates.
               direction is a string ('N', 'S', 'E', 'W') indicating the current direction.
    """
    x, y = 0, 0
    directions = ['N', 'E', 'S', 'W']
    dir_index = 0

    for command in commands:
        if command == 'L':
            dir_index = (dir_index - 1) % 4
        elif command == 'R':
            dir_index = (dir_index + 1) % 4
        else:
            if command in ['F', '^']:
                y += 1
            elif command in ['v']:
                y -= 1
            elif command in ['<']:
                x -= 1
            elif command in ['>']:
                x += 1

    return (x, y, directions[dir_index])
