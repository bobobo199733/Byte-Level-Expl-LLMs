def min_window_substring(s, t):
    result = None
    window = {}
    need = {}
    for char in t:
        if char not in need:
            need[char] = 0
        need[char] += 1

    while end < len(s):

                window[char] = 0
            window[char] += 1
            if char in need:
                count -= 1 if char in need and window[char] == need[char] else 0
            if count == 0:

                if newlen < minlen:
                    result = s[start:end+1]
