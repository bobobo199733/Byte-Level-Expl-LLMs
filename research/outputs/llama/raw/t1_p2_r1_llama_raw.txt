```python
def roman_to_int(s: str) -> int:
    """
    Converts Roman numerals to integers.
    
    Parameters:
    s (str): The Roman numeral string.
    
    Returns:
    int: The integer value of the Roman numerals.

    Raises:
    ValueError: If the input contains invalid characters or formatting.
    """

    roman_map = {
        'I': 1, 'V': 5, 'X': 10, 'L': 50, 'C': 100, 'D': 500, 'M': 1000
    }

    # Initialize result variable to store the final integer value.
    result = 0

    # Check for invalid input (empty string or none of the expected characters).
    if not s:
        raise ValueError("Input must be a non-empty string.")

    # Iterate over each character in the input string.
    for i, char in enumerate(s):
        try:
            # Try to get the value from the roman_map dictionary. If it's not found, it will raise KeyError.
            num = roman_map[char.upper()]
            
            # Check if we're dealing with subtractive notation (e.g., IV). 
            # If this is true, then we add the value of the current numeral to the result only if it's greater than the previous one.
            if i > 0 and num > roman_map[s[i - 1].upper()]:
                raise ValueError("Invalid subtractive notation.")
            
            result += num
            
        except KeyError:
            # If a character is not in the roman_map dictionary, we raise an error with a custom message.
            raise ValueError(f"Invalid input: '{char}' is not a valid Roman numeral.")

    return result
```
