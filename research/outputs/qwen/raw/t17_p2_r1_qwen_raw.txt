Sure! To implement an LRU (Least Recently Used) cache in Python, we can use two data structures:
1. A dictionary to store the actual cache values and their timestamps.
2. A doubly linked list to maintain the order of usage.

Here is a complete implementation:

```python
class Node:
    def __init__(self, key, value):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None

class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}  # Maps keys to nodes in the doubly linked list
        self.head = Node(0, 0)  # Dummy head node
        self.tail = Node(0, 0)  # Dummy tail node
        self.head.next = self.tail
        self.tail.prev = self.head

    def _add_node(self, node):
        """Add a new node right after the head."""
        next_node = self.head.next
        self.head.next = node
        node.prev = self.head
        node.next = next_node
        next_node.prev = node

    def _remove_node(self, node):
        """Remove an existing node from the linked list."""
        prev_node = node.prev
        next_node = node.next
        prev_node.next = next_node
        next_node.prev = prev_node

    def _move_to_head(self, node):
        """Move a specified node in-between to the head position."""
        self._remove_node(node)
        self._add_node(node)

    def _pop_tail(self):
        """Remove and return the node at the tail. This is the least recently used key."""
        res = self.tail.prev
        self._remove_node(res)
        return res

    def get(self, key):
        if key in self.cache:
            # If the key exists, move it to the head.
            self._move_to_head(self.cache[key])
            return self.cache[key].value
        return -1  # Return -1 if the key is not found.

    def put(self, key, value):
        if key in self.cache:
            # If the key exists, remove and update its value.
            old_node = self.cache[key]
            self._remove_node(old_node)
        new_node = Node(key, value)
        self.cache[key] = new_node
        self._add_node(new_node)

        if len(self.cache) > self.capacity:
            # If the cache exceeds capacity, remove the least recently used item.
            tail_node = self._pop_tail()
            del self.cache[tail_node.key]

def lru_cache(operations, capacity):
    cache = LRUCache(capacity)
    results = []
    for operation in operations:
        if operation[0] == "get":
            result = cache.get(operation[1])
            results.append(result)
        elif operation[0] == "put":
            cache.put(operation[1], operation[2])
    return results

# Example usage
operations = [
    ("put", 1, 1),
    ("put", 2, 2),
    ("get", 1),
    ("put", 3, 3),  # Evicts key 2
    ("get", 2),
    ("put", 4, 4),  # Evicts key 1
    ("get", 1),
    ("get", 3),
    ("get", 4)
]
capacity = 3
print(lru_cache(operations, capacity))
```

### Explanation:
- **Node Class**: Represents each item in the cache. Each node contains a `key`, `value`, and pointers to its `prev` and `next` nodes.
- **LRUCache Class**:
  - `_add_node`: Adds a new node right after the head.
  - `_remove_node`: Removes an existing node from the linked list.
  - `_move_to_head`: Moves a specified node in-between to the head position (most recently used).
  - `_pop_tail`: Removes and returns the node at the tail, which is the least recently used key.
  - `get(key)`: Retrieves a value by key and moves the corresponding node to the front of the list if it exists.
  - `put(key, value)`: Adds or updates a key-value pair. If the cache exceeds capacity, evicts the least recently used item.

### Usage:
- The function `lru_cache` processes a list of operations (`operations`) and returns the results for all "get" operations as specified in the problem requirements.
