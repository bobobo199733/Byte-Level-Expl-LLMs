def sieve_primes(n):
    """
    Returns a list of all prime numbers up to n using the Sieve of Eratosthenes algorithm.

    Args:
        n (int): The upper limit for finding prime numbers.

    Returns:
        list: A list of prime numbers up to n.
    """
    
    # Initialize a boolean array, prime, of size n+1. We'll use this array to mark composite numbers.
    prime = [True] * (n + 1)
    
    # 0 and 1 are not prime numbers
    prime[0] = prime[1] = False
    
    # Starting from the first prime number, 2, iterate over the array.
    for p in range(2, int(n ** 0.5) + 1):
        
        # If p is a prime number, mark its multiples as composite
        if prime[p]:
            for i in range(p * p, n + 1, p):
                prime[i] = False
    
    # Return a list of all prime numbers in the array
    return [p for p in range(2, n + 1) if prime[p]]
