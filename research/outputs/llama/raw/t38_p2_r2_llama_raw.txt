def expression_to_postfix(expression):
    """
    Convert a mathematical infix expression into postfix notation.

    Args:
        expression (str): A string containing a mathematical infix expression.

    Returns:
        str: The equivalent postfix notation of the input expression.

    Example:
        >>> expression_to_postfix('A + B * C')
        'ABC*+'
    """

    # Define operator precedence
    precedence = {'+': 1, '-': 1, '*': 2, '/': 2}

    # Initialize output queue and operators stack
    output_queue = []
    operators_stack = []

    # Iterate through characters in the input expression
    for char in expression:
        # Ignore whitespace
        if char == ' ':
            continue

        # Check if character is a digit or uppercase letter (operand)
        if char.isalnum():
            output_queue.append(char)

        # Check if character is an operator
        elif char in precedence:
            # Pop operators from stack with higher or equal precedence and push to output queue
            while operators_stack and operators_stack[-1] in precedence and precedence[operators_stack[-1]] >= precedence[char]:
                output_queue.append(operators_stack.pop())
            # Push current operator onto the stack
            operators_stack.append(char)

        # Check if character is a left parenthesis (LP)
        elif char == '(':
            # Push LP onto the stack
            operators_stack.append(char)

        # Check if character is a right parenthesis (RP)
        elif char == ')':
            # Pop operators from stack until we find the matching LP and push to output queue
            while operators_stack[-1] != '(':
                output_queue.append(operators_stack.pop())
            # Discard LP from the stack
            operators_stack.pop()

    # Pop any remaining operators from the stack and append them to the output queue
    while operators_stack:
        output_queue.append(operators_stack.pop())

    # Join characters in the output queue into a string and return
    return ''.join(output_queue)


# Test the function with some examples
print(expression_to_postfix('A + B * C'))  # Output: ABC*+
print(expression_to_postfix('( A + B ) * ( C - D )'))  # Output: AB+C+D-C**
