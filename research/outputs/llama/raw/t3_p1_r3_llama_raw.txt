```python
def evaluate(expr: str) -> int:
    """
    Evaluates an arithmetic expression and returns the integer result.

    Args:
        expr (str): The expression to evaluate, ignoring whitespace.

    Returns:
        int: The integer result of the evaluated expression.

    Raises:
        ValueError: If the expression is invalid.
    """

    # Define a dictionary mapping operator characters to their corresponding functions
    operators = {
        '+': lambda x, y: x + y,
        '-': lambda x, y: x - y,
        '*': lambda x, y: x * y,
        '/': lambda x, y: int(x / y) if isinstance(x, int) and isinstance(y, int) else x / y
    }

    # Initialize a stack to store intermediate results
    stack = []

    # Initialize an empty expression string
    curr_expr = ''

    # Iterate over each character in the input expression
    i = 0
    while i < len(expr):
        char = expr[i]

        # If the character is a digit or a decimal point, append it to the current expression
        if char.isdigit() or char == '.':
            curr_expr += char

        # If the character is an operator, push the current expression onto the stack and reset it
        elif char in operators:
            if not stack:
                raise ValueError("Invalid expression: missing operand")
            num = int(curr_expr) if curr_expr else stack.pop()
            func = operators[char]
            if expr[i + 1] == '(':
                # Handle function application (e.g. `sin(x)` -> `sin`, `x`)
                i += 2
                arg = evaluate(expr[i:])
                i = len(expr)
                num = func(num, arg)
            else:
                op = stack.pop()
                if isinstance(op, str):
                    raise ValueError("Invalid expression: unsupported operator")
                num = func(op, num)
            stack.append(num)
            curr_expr = ''
        # If the character is a left parenthesis, push it onto the stack
        elif char == '(':
            stack.append(char)

        # If the character is a right parenthesis, evaluate any nested subexpressions
        elif char == ')':
            if not stack or stack[-1] != '(':
                raise ValueError("Invalid expression: mismatched parentheses")
            while stack and isinstance(stack[-1], int) or (isinstance(stack[-1], str) and stack[-1] in operators):
                num = stack.pop()
                if isinstance(num, str):
                    func = operators[num]
                else:
                    func = lambda x, y: x + y
                op = stack.pop() if stack else 0
                num = func(op, num)
                stack.append(num)

        # If the character is not a digit, operator, or parenthesis, raise an error
        else:
            raise ValueError(f"Invalid expression: unexpected character '{char}'")

        i += 1

    # Evaluate any remaining expressions in the stack
    while stack and isinstance(stack[-1], int) or (isinstance(stack[-1], str) and stack[-1] in operators):
        num = stack.pop()
        if isinstance(num, str):
            func = operators[num]
        else:
            func = lambda x, y: x + y
        op = stack.pop() if stack else 0
        num = func(op, num)
        stack.append(num)

    # Check for missing end-of-expression marker
    if stack and isinstance(stack[-1], int):
        raise ValueError("Invalid expression: unexpected trailing digits")

    # Return the final result
    return stack[0]
```
This function uses a stack to store intermediate results and supports integers, addition, subtraction, multiplication, division (truncating toward zero), and parentheses. It raises `ValueError` for invalid expressions or unsupported operators.
