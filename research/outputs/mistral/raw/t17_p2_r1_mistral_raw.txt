I'll provide you with a Python function that simulates an LRU cache based on your requirements. The function uses a doubly linked list to maintain the order of recently used items and a dictionary for fast access.

```python
class Node:
    def __init__(self, key, value):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None

class LRUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = {}
        self.head = None
        self.tail = None

    def get(self, key):
        if key not in self.cache:
            return None

        # Move accessed node to the head of the list (most recently used)
        node = self.cache[key]
        if self.head != node:
            if self.head:
                self.head.prev = node.prev
            else:
                self.tail = node
            if node.next:
                node.next.prev = node.prev
            else:
                self.head = node
            node.prev, node.next = self.head, self.head.next
            self.head.next = node

        return self.cache[key]['value']

    def put(self, key, value):
        if key in self.cache:
            # Update the value and move it to the head of the list (most recently used)
            self.cache[key]['value'] = value
            node = self.cache[key]
            if self.head != node:
                if self.head:
                    self.head.prev = node.prev
                else:
                    self.tail = node
                if node.next:
                    node.next.prev = node.prev
                else:
                    self.head = node
                node.prev, node.next = self.head, self.head.next
                self.head.next = node
        else:
            # Add a new node to the cache if it's not full
            if len(self.cache) >= self.capacity:
                # Evict the least recently used key (the one at the tail of the list)
                evicted_key = self.tail.key
                del self.cache[evicted_key]
                if self.head == self.tail:
                    self.head = None
                else:
                    self.tail = self.tail.prev
                    self.tail.next = None

            # Add the new key-value pair and its corresponding node to the cache and list
            self.cache[key] = {'value': value}
            node = Node(key, value)
            if not self.head:
                self.head = node
                self.tail = node
            else:
                node.next = self.head
                self.head.prev = node
                self.head = node

def lru_cache(operations, capacity):
    cache = LRUCache(capacity)
    results = []
    for operation in operations:
        if operation[0] == 'put':
            cache.put(operation[1], operation[2])
        elif operation[0] == 'get':
            results.append(cache.get(operation[1]))
    return results
```

Now you can use the `lru_cache` function to simulate an LRU cache with given operations and capacity.

For example:

```python
operations = [("put", 1, 1), ("put", 2, 2), ("get", 1), ("put", 3, 3), ("get", 2), ("get", 3), ("put", 4, 4), ("get", 1), ("get", 2)]
capacity = 2
lru_cache(operations, capacity)
# Output: [1, 2, None, 2, 3, None]
```
