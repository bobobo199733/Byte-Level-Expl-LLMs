import math

def longest_unique_substring(s):
    """
    Returns the length of the longest substring without repeating characters.

    Args:
        s (str): Input string.

    Returns:
        int: Length of the longest unique substring.
    """
    if not s:  # handle empty strings
        return 0

    char_set = set()  # keep track of seen characters
    left = 0  # left pointer of sliding window
    max_len = 0  # maximum length found so far

    for right, c in enumerate(s):  # iterate over string with right pointer
        while c in char_set:  # if character is repeated, shrink window from left
            char_set.remove(s[left])

        char_set.add(c)  # add new character to set
        max_len = max(max_len, right - left + 1)

    return max_len
