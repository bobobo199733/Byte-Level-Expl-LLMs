def fast_power(base, exponent):
    if exponent == 0:
        return 1
    elif exponent < 0:
        raise ValueError("Exponent must be non-negative")
    result = 1
    while exponent > 0:
        if exponent % 2 == 1:
            result *= base
        base *= base
        exponent //= 2
    return result
