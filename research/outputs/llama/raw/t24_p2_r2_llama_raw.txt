def kth_largest(nums, k):
    """
    Returns the kth largest value from nums.

    Args:
        nums (list): A list of integers.
        k (int): The position of the number to be found (1-indexed).

    Returns:
        int: The kth largest value in nums.

    Raises:
        ValueError: If k is invalid (less than 1 or greater than the length of nums).
    """
    if not isinstance(nums, list) or not all(isinstance(x, int) for x in nums):
        raise TypeError("nums must be a list of integers")
    if not isinstance(k, int) or k < 1:
        raise ValueError("k must be a positive integer")

    sorted_nums = sorted(nums, reverse=True)
    if k > len(sorted_nums):
        raise ValueError(f"Invalid k. There are only {len(sorted_nums)} unique values.")
    return sorted_nums[k - 1]
