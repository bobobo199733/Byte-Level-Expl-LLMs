import heapq

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next
    
    def __lt__(self, other):
        # Define comparison based on node values for heapq to work correctly
        return self.val < other.val

def merge_k_sorted_lists(lists):
    if not lists:
        return None
    
    # Use a min heap to store the nodes from all lists
    heap = []
    
    # Initialize the heap with the first node of each list
    for lst in lists:
        if lst:  # Ensure there's at least one element in the list
            heapq.heappush(heap, lst)
    
    dummy = ListNode()  # Dummy head to simplify the result construction
    current = dummy
    
    while heap:
        # Pop the smallest node from the heap
        node = heapq.heappop(heap)
        
        # Add the next node of the popped node (if any) to the heap
        if node.next:
            heapq.heappush(heap, node.next)
        
        # Append the current node's value to the result list
        current.next = node
        current = current.next
    
    return dummy.next

def create_linked_list(elements):
    if not elements:
        return None
    head = ListNode(elements[0])
    current = head
    for element in elements[1:]:
        current.next = ListNode(element)
        current = current.next
    return head

# Create multiple sorted linked lists
list1 = create_linked_list([1, 4, 5])
