def decode_string(s: str) -> str:
    if not s or len(s) == 0:
        raise ValueError("Invalid input")

    stack = []
    current_num = ""
    current_str = ""
    num = 0

    for char in s:
        if char.isdigit():
            current_num += char
        elif char.isalpha():
            current_str += char
        elif char == '[':
            stack.append((current_str, num))
            current_str, num = "", 0
        elif char == ']':
            if not stack:
                raise ValueError("Invalid input")
            prev_str, prev_num = stack.pop()
            current_str = prev_str + current_num * prev_num + current_str
            num = 0

    if stack:
        raise ValueError("Invalid input")

    return current_str
