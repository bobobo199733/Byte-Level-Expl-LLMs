def binary_search_rotated(values, target):

        if values[mid] == target:
            return mid

        if values[low] <= values[mid]:
            if target < values[mid] and target >= values[low]:
                high = mid - 1
            else:
                low = mid + 1

        # If the right half of the array is sorted, then the target should be on the right side.
        else:
            if target > values[mid] and target <= values[high]:
