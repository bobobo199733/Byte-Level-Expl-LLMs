def min_window_substring(s, t):
    from collections import Counter
    need = Counter(t)
    missing = len(t)
    i = start = end = 0
    for j in range(len(s)):
        if need[s[j]] > 0:
            missing -= 1
        need[s[j]] -= 1
        if not missing:
            while i < j and need[s[i]] < 0:
                need[s[i]] += 1
                i += 1
            if not end or j - i + 1 < end - start + 1:
                start, end = i, j
    return s[start:end+1]
