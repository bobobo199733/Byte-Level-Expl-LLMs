```python
def build_max_heap(values):
    n = len(values)
    for i in range(n // 2 - 1, -1, -1):
        heapify_down(i, n, values)

def heapify_down(i, n, values):
    largest = i
    left = 2 * i + 1
    right = 2 * i + 2

    if left < n and values[left] > values[largest]:
        largest = left

    if right < n and values[right] > values[largest]:
        largest = right

    if largest != i:
        values[i], values[largest] = values[largest], values[i]
        heapify_down(largest, n, values)

def heap_sort(values):
    if not values:
        return []

    build_max_heap(values)

    sorted_values = []
    for i in range(len(values) - 1, -1, -1):
        sorted_values.append(values[0])
        values[0] = values[i]
        del values[i]
        heapify_down(0, len(values), values)

    return sorted_values
```

This implementation builds a max-heap using the `build_max_heap()` function and then sorts the list by iteratively removing the root (the largest element) and rebuilding the heap with the rest of the elements. The original input list is not modified, and duplicate values are preserved.
