        n = len(s)
        dp = [[False] * n for _ in range(n)]
        max_length, center = (0, 0)
        for i in range(n):
            dp[i][i] = True
        for l in range(2, n + 1):

                        center = i

        valid = {'I', 'V', 'X', 'L', 'C', 'D', 'M'}
        subtractive = {('IV', 4), ('IX', 9), ('XL', 40), ('XC', 90), ('CD', 400), ('CM', 900)}

        for char in s:

            char, num = (None, 0)

                    break
