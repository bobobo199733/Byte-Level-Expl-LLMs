def is_valid(board, row, col, num):
    # Check if we can put the number in the current position (row, col)
    for x in range(9):
        # Check if there is any other occurrence of the same number in the row.
        if board[row][x] == num:
            return False
        
        # Check if there is any other occurrence of the same number in the column.
        if board[x][col] == num:
            return False
    
    # Calculate the starting point of the 3x3 box
    start_row = row - row % 3
    start_col = col - col % 3
    
    for i in range(3):
        for j in range(3):
            if board[i + start_row][j + start_col] == num:
                return False
    
    return True

def find_empty(board):
    # Find the next empty cell (0) on the board
    for row in range(len(board)):
        for col in range(len(board[0])):
            if board[row][col] == 0:
                return row, col
    return None, None  # If no empty cell is found

def sudoku_solver(board):
    # Find an empty cell
    row, col = find_empty(board)
    
    # If there are no more empty cells, the puzzle is solved.
    if row is None and col is None:
        return True
    
    for num in range(1, 10):  # Try numbers from 1 to 9
        if is_valid(board, row, col, num):
            board[row][col] = num
            
            if sudoku_solver(board):
                return True
            
            # Backtrack and reset the cell to 0
            board[row][col] = 0
    
    # If no number can be placed in this cell, raise an error.
    raise ValueError("The puzzle is unsolvable or invalid.")
