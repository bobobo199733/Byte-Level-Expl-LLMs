def sieve_primes(n):
    """
    Returns a list of all prime numbers less than or equal to n.

    Args:
        n (int): The upper bound for prime number generation.

    Returns:
        list: A list of prime numbers.
    """

    # Create a boolean array, prime, of size n+1
    prime = [True] * (n + 1)
    
    # 0 and 1 are not prime numbers
    prime[0] = prime[1] = False
    
    # Iterate over the array starting from 2
    for p in range(2, int(n ** 0.5) + 1):
        # If p is a prime number, mark its multiples as non-prime
        if prime[p]:
            for i in range(p * p, n + 1, p):
                prime[i] = False
    
    # Return a list of all prime numbers
    return [p for p in range(2, n + 1) if prime[p]]
