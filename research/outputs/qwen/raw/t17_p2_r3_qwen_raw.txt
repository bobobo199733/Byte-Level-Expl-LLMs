Certainly! To implement an LRU (Least Recently Used) cache in Python without using any external libraries, we can use a combination of a dictionary and a doubly linked list. The dictionary will allow us to quickly access the nodes by their keys, while the doubly linked list will help maintain the order of usage.

Here's how you can implement this:

```python
class Node:
    def __init__(self, key, value):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None

class LRUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = {}  # Maps keys to nodes
        # Initialize a dummy head and tail node for the doubly linked list
        self.head = Node(0, 0)
        self.tail = Node(0, 0)
        self.head.next = self.tail
        self.tail.prev = self.head

    def get(self, key: int) -> int:
        if key in self.cache:
            node = self.cache[key]
            # Move the accessed node to the head of the list
            self._remove(node)
            self._add(node)
            return node.value
        return -1  # Key not found

    def put(self, key: int, value: int) -> None:
        if key in self.cache:
            # If the key exists, remove the existing node first
            node = self.cache[key]
            self._remove(node)
        new_node = Node(key, value)
        self.cache[key] = new_node
        self._add(new_node)
        
        # Check if we have exceeded the capacity
        if len(self.cache) > self.capacity:
            # Evict the least recently used node (node at tail.prev)
            lru_node = self.tail.prev
            self._remove(lru_node)
            del self.cache[lru_node.key]

    def _add(self, node):
        # Add a new node to the head of the list
        first_node = self.head.next
        self.head.next = node
        node.prev = self.head
        node.next = first_node
        first_node.prev = node

    def _remove(self, node):
        # Remove an existing node from the doubly linked list
        prev_node = node.prev
        next_node = node.next
        prev_node.next = next_node
        next_node.prev = prev_node

def lru_cache(operations, capacity):
    cache = LRUCache(capacity)
    results = []
    
    for operation in operations:
        if operation[0] == "get":
            result = cache.get(operation[1])
            results.append(result)
        elif operation[0] == "put":
            cache.put(operation[1], operation[2])
    
    return results

# Example usage
operations = [("put", 1, 1), ("put", 2, 2), ("get", 1), ("put", 3, 3), ("get", 2), ("get", 3)]
capacity = 2
print(lru_cache(operations, capacity))
```

### Explanation:

1. **Node Class**: Represents each node in the doubly linked list.
2. **LRUCache Class**:
   - Contains a dictionary (`cache`) to map keys to their respective nodes.
   - Maintains dummy head and tail nodes for easy insertion and removal operations.
   - Provides `get` and `put` methods to handle cache operations.
3. **get Method**: Retrieves the value of a key from the cache, moves it to the front (head) of the list if found.
4. **put Method**: Inserts or updates a key-value pair in the cache, ensuring that the capacity is not exceeded by evicting the least recently used item.
5. **_add and _remove Methods**: Helper methods to insert and remove nodes from the doubly linked list.

The `lru_cache` function processes each operation in `operations`, calling either `get` or `put` as appropriate, and collects results for all get operations into a list which is returned at the end.
