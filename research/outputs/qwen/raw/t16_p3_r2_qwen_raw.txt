def sudoku_solver(board):
    def is_valid(num, row, col):
        # Check the row for any duplicate numbers
        for i in range(9):
            if board[row][i] == num:
                return False
        
        # Check the column for any duplicate numbers
        for i in range(9):
            if board[i][col] == num:
                return False

        # Check the 3x3 sub-grid for any duplicate numbers
        box_row = (row // 3) * 3
        box_col = (col // 3) * 3
        for i in range(3):
            for j in range(3):
                if board[box_row + i][box_col + j] == num:
                    return False
        
        return True

    def solve():
        # Find the next empty cell
        for row in range(9):
            for col in range(9):
                if board[row][col] == 0:
                    # Try values from 1 to 9
                    for num in range(1, 10):
                        if is_valid(num, row, col):
                            # Place the number and move forward
                            board[row][col] = num
                            if solve():
                                return True
                            else:
                                # If placing the number leads to a dead end, backtrack
                                board[row][col] = 0
                    # If no valid number can be placed, return False (backtrack)
                    return False
        
        # If all cells are filled, the Sudoku is solved
        return True

    if solve():
        return board
    else:
        print("No solution exists.")
        return None
