def robot_simulator(commands):
    """
    Simulate a robot moving on a grid based on given commands.

    Args:
        commands (list): A list of strings representing commands to move the robot.
                         'L' or 'R' to turn, and 'F' or 'B' to move forward or backward respectively.

    Returns:
        tuple: The final position (x, y) and direction ('N', 'S', 'E', 'W') of the robot.
    """

    # Initialize grid size to 4x4
    grid_size = 4

    # Initialize starting position at the center of the grid
    x, y = grid_size // 2, grid_size // 2

    # Initialize direction as North
    directions = ['N', 'S', 'E', 'W']
    direction_index = 0

    for command in commands:
        if command == 'L':
            # Turn left by decrementing the index and taking modulus with 4 to stay within bounds
            direction_index = (direction_index - 1) % 4
        elif command == 'R':
            # Turn right by incrementing the index and taking modulus with 4 to stay within bounds
            direction_index = (direction_index + 1) % 4
        elif command == 'F':
            # Move forward based on current direction
            if directions[direction_index] in ['N', 'S']:
                y += (-1 if directions[direction_index] == 'N' else 1)
            else:
                x += (-1 if directions[direction_index] == 'W' else 1)
        elif command == 'B':
            # Move backward based on current direction
            if directions[direction_index] in ['N', 'S']:
                y -= (-1 if directions[direction_index] == 'N' else 1)
            else:
                x -= (-1 if directions[direction_index] == 'W' else 1)

    return (x, y), directions[direction_index]
