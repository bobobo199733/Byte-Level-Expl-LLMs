def remove_k_digits(num: str, k: int) -> str:
    stack = []
    for n in num:
        while k > 0 and len(stack) > 0 and stack[-1] > n:
            stack.pop()
            k -= 1
        stack.append(n)
    if k != 0:
        stack = stack[:-k]
    return '0' if not stack else ''.join(stack).lstrip('0') or '0'
