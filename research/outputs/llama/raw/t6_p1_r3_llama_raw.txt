def wildcard_match(s: str, p: str) -> bool:
    """
    Return True if the entire string s matches the wildcard pattern p, otherwise return False.

    Rules:
    - '?' matches exactly one character
    - '*' matches any sequence of characters, including the empty sequence
    - Matching must cover the entire string
    """

    # Handle edge case where both strings are empty
    if len(p) == 0 and len(s) > 0:
        return False
    
    # If the pattern is empty but the string isn't, they don't match
    if len(p) == 0 and len(s) != 0:
        return False

    # Initialize a table to store results of subproblems
    dp = [[False] * (len(p) + 1) for _ in range(len(s) + 1)]

    # Empty string matches empty pattern
    dp[0][0] = True

    # '*' can match an empty string, so the first row should all be True
    for j in range(1, len(p) + 1):
        if p[j - 1] == '*':
            dp[0][j] = dp[0][j - 1]

    # Fill in the rest of the table
    for i in range(1, len(s) + 1):
        for j in range(1, len(p) + 1):
            if p[j - 1] == '*':
                # '*' can match any sequence of characters, including the empty string
                dp[i][j] = dp[i][j - 1] or dp[i - 1][j]
            elif p[j - 1] == '?':
                # '?' matches exactly one character
                dp[i][j] = dp[i - 1][j - 1]
            else:
                # If the current characters match, check if we can get here from a smaller subproblem
                dp[i][j] = dp[i - 1][j - 1] and s[i - 1] == p[j - 1]

    # The result is stored in the bottom-right corner of the table
    return dp[len(s)][len(p)]
