def is_valid_move(board, row, col, num):
    # Check if `num` is not already placed in current row, column, and 3x3 box
    for x in range(9):
        if board[row][x] == num:
            return False
        if board[x][col] == num:
            return False
    
    start_row = row - row % 3
    start_col = col - col % 3
    for i in range(3):
        for j in range(3):
            if board[i + start_row][j + start_col] == num:
                return False

    return True

def find_empty_location(board):
    # Find an empty cell (0) on the board
    for row in range(9):
        for col in range(9):
            if board[row][col] == 0:
                return row, col
    return None

def sudoku_solver(board):
    empty_loc = find_empty_location(board)
    
    # If there is no empty cell, we have solved the puzzle
    if not empty_loc:
        return True
    
    row, col = empty_loc
    
    for num in range(1, 10):  # Loop over numbers from 1 to 9
        if is_valid_move(board, row, col, num):
            board[row][col] = num
            
            if sudoku_solver(board):
                return True
            
            # If placing `num` doesn't lead to a solution, backtrack and reset the cell
            board[row][col] = 0
    
    # No valid number can be placed in this empty location. The puzzle is unsolvable.
    raise ValueError("Unsolvable Sudoku puzzle")

2. **find_empty_location**: This function finds an empty cell (represented by 0) on the board. It returns None if there are no more empty cells.
3. **sudoku_solver**:

   - Recursively calls itself for the next cell. If placing a number does not lead to a solution, it backtracks by resetting the cell to 0.
