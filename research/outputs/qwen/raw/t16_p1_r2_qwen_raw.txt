```python
def is_valid(board, row, col, num):
    # Check if we can place 'num' in the current board[row][col]
    for x in range(9):
        if board[row][x] == num:
            return False  # Same number already exists in the row

    for y in range(9):
        if board[y][col] == num:
            return False  # Same number already exists in the column

    start_row = row - row % 3
    start_col = col - col % 3
    for i in range(3):
        for j in range(3):
            if board[i + start_row][j + start_col] == num:
                return False  # Same number already exists in the 3x3 sub-box

    return True

def solve_sudoku(board):
    def backtrack():
        nonlocal board
        for row in range(9):
            for col in range(9):
                if board[row][col] == '.':
                    for num in '123456789':
                        if is_valid(board, row, col, num):
                            # Place the number on the current empty cell
                            board[row][col] = num

                            # Continue to fill the rest of the cells
                            if backtrack():
                                return True

                            # If placing 'num' didn't lead to a solution, undo it
                            board[row][col] = '.'
                    return False  # If no number can be placed in this cell, backtrack
        return True  # All cells are filled

    # Convert '.' to '0' for easier comparison and conversion back
    for row in range(9):
        for col in range(9):
            if board[row][col] == '.':
                board[row][col] = '0'

    solve_sudoku_helper(board)

    # Convert '0' back to '.'
    for row in range(9):
        for col in range(9):
            if board[row][col] == '0':
                board[row][col] = '.'

def sudoku_solver(board):
    return solve_sudoku(board)
```

### Explanation:
1. **is_valid Function**: This function checks whether it is valid to place a number `num` at the given position `(row, col)` on the Sudoku board.
2. **solve_sudoku Helper Function**: This recursive helper function tries to fill each empty cell with numbers from 1 to 9 and uses backtracking to find a solution.
3. **sudoku_solver Function**: Converts the input list of strings (where '.' represents an empty cell) to a list of integers for easier processing, calls the `solve_sudoku` helper, and converts it back.
